' == Activity: UC01 Login ==
@startuml
title Activity - UC01 Login
|Pengguna|
start
:Membuka halaman Login;
:Memasukkan email dan password;
:Menekan tombol LOGIN;
|Sistem QAYRA|
:Memverifikasi email dan password;
if (Kredensial valid & akun aktif?) then ([ya])
  :Menyimpan sesi pengguna;
  :Mengarahkan ke Dashboard sesuai peran;
else ([tidak])
  :Menampilkan pesan "Email atau password salah";
  stop
endif
|Pengguna|
:Melihat Dashboard;
stop
@enduml

' == Activity: UC02 Logout ==
@startuml
title Activity - UC02 Logout
|Pengguna|
start
:Menekan ikon Logout;
:Mengonfirmasi keluar aplikasi;
if (Konfirmasi keluar?) then ([ya])
  |Sistem QAYRA|
  :Menghapus sesi pengguna;
  :Mengarahkan ke halaman Login;
  |Pengguna|
  :Melihat halaman Login;
  stop
else ([tidak])
  |Pengguna|
  :Tetap berada di aplikasi;
  stop
endif
@enduml

' == Activity: UC03 Melakukan Analisis Kulit Wajah ==
@startuml
title Activity - UC03 Melakukan Analisis Kulit Wajah
|Client|
start
:Membuka halaman Kamera;
:Mengambil foto wajah atau mengunggah gambar;
:Meninjau pratinjau foto;
if (Foto sesuai?) then ([ya])
  :Mengonfirmasi foto;
  |Sistem QAYRA|
  :Memproses gambar dengan model MobileNetV2;
  :Menghasilkan kondisi kulit & rekomendasi;
  :Menyimpan hasil analisis;
  |Client|
  :Melihat halaman Detail Hasil Analisis;
  stop
else ([tidak])
  :Mengambil ulang foto;
  stop
endif
@enduml

' == Activity: UC09 Mengelola Data User ==
@startuml
title Activity - UC09 Mengelola Data User
|Admin|
start
:Membuka halaman Manajemen User;
:Memilih aksi pengelolaan;
switch (Aksi?)
case ([Tambah])
  :Mengisi form data user baru;
case ([Edit])
  :Mengubah data user terpilih;
case ([Hapus])
  :Mengonfirmasi penghapusan user;
endswitch
|Sistem QAYRA|
:Memvalidasi & menyimpan perubahan data user;
if (Operasi berhasil?) then ([ya])
  :Memperbarui daftar user;
  |Admin|
  :Melihat daftar user terbaru;
  stop
else ([tidak])
  :Menampilkan pesan kesalahan;
  |Admin|
  stop
endif
@enduml

' == Activity: UC10 Mengelola Data Analisis ==
@startuml
title Activity - UC10 Mengelola Data Analisis
|Admin|
start
:Membuka halaman Hasil Analisis;
:Mencari / menelusuri data klien;
:Memilih aksi (Detail atau Hapus);
if (Aksi Hapus?) then ([ya])

  :Menghapus data analisis terpilih;

  stop
else ([tidak / Detail])
  |Sistem QAYRA|
  :Membuka Detail Hasil Analisis;
  |Admin|
  :Meninjau hasil analisis;
  stop
endif
@enduml
